#include <iostream>
#include <string>
using namespace std;

string xorOperation(string a, string b)
{
    string result = "";

    for (int i = 1; i < b.length(); i++)
    {
        if (a[i] == b[i])
            result += '0';
        else
            result += '1';
    }

    return result;
}

string mod2Division(string data, string divisor)
{
    int divisorLength = divisor.length();
    string temp = data.substr(0, divisorLength);

    int pos = divisorLength;

    while (pos < data.length())
    {
        if (temp[0] == '1')
            temp = xorOperation(divisor, temp);
        else
            temp = xorOperation(string(divisorLength, '0'), temp);

        temp += data[pos];
        pos++;
    }

    if (temp[0] == '1')
        temp = xorOperation(divisor, temp);
    else
        temp = xorOperation(string(divisorLength, '0'), temp);

    return temp;
}

string charToBinary(string data)
{
    string binary = "";

    for (char c : data)
    {
        int value = (int)c;

        for (int i = 7; i >= 0; i--)
        {
            binary += ((value >> i) & 1) ? '1' : '0';
        }
    }

    return binary;
}

void calculateCRC(string data, string polynomial, string name)
{
    string binaryData = charToBinary(data);

    int degree = polynomial.length() - 1;

    string appendedData = binaryData + string(degree, '0');

    string remainder = mod2Division(appendedData, polynomial);

    cout << name << " : ";
    
    for (char bit : remainder)
        cout << bit;

    cout << endl;
}

int main()
{
    string data;

    cout << "Enter the data: ";
    cin >> data;

    cout << "\nCRC Results\n";
    cout << "-----------\n";

    // CRC-12 polynomial:
    // x^12 + x^11 + x^3 + x^2 + x + 1
    calculateCRC(
        data,
        "1100000001111",
        "CRC-12"
    );

    // CRC-16 polynomial:
    // x^16 + x^15 + x^2 + 1
    calculateCRC(
        data,
        "11000000000000101",
        "CRC-16"
    );

    // CRC-CCITT polynomial:
    // x^16 + x^12 + x^5 + 1
    calculateCRC(
        data,
        "10001000000100001",
        "CRC-CCITT"
    );

    return 0;
}
