#include <iostream>
using namespace std;

#define INF 999

int main()
{
    int n, source;
    int cost[10][10], dist[10], visited[10] = {0};

    cout << "Enter number of vertices: ";
    cin >> n;

    cout << "Enter cost matrix:\n";
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            cin >> cost[i][j];

    cout << "Enter source vertex: ";
    cin >> source;
    source--;

    for (int i = 0; i < n; i++)
        dist[i] = cost[source][i];

    dist[source] = 0;
    visited[source] = 1;

    for (int count = 1; count < n; count++)
    {
        int min = INF, u = -1;

        for (int i = 0; i < n; i++)
            if (!visited[i] && dist[i] < min)
            {
                min = dist[i];
                u = i;
            }

        if (u == -1) break;

        visited[u] = 1;

        for (int v = 0; v < n; v++)
            if (!visited[v] &&
                dist[u] + cost[u][v] < dist[v])
                dist[v] = dist[u] + cost[u][v];
    }

    cout << "\nShortest distances:\n";
    for (int i = 0; i < n; i++)
        cout << "From " << source + 1 << " to "
             << i + 1 << " = " << dist[i] << endl;

    return 0;
}
